### System Context

```mermaid
C4Context
    title Контекст системи «Поштомат громади»

    Person(resident, "Мешканець", "Отримує SMS із кодом і забирає посилку на поштоматі")
    Person(delivery, "Служба доставки", "Реєструє посилки через сканер")
    Person(staff, "Персонал громади", "Переглядає список посилок")
    Person(monitor, "Моніторинг", "Перевіряє стан сервісу")

    System(locker, "Поштомат громади", "Приймає дані про посилки та допомагає мешканцям їх отримати")
    System_Ext(sms, "SMS-провайдер", "Надсилає мешканцям SMS")

    Rel(delivery, locker, "Реєструє посилки")
    Rel(resident, locker, "Вводить код отримання")
    Rel(locker, resident, "Надсилає код у SMS")
    Rel(locker, sms, "Замовляє надсилання SMS")
    Rel(staff, locker, "Переглядає посилки")
    Rel(monitor, locker, "Перевіряє стан сервісу")
```

### Container

```mermaid
C4Container
    title Контейнери системи «Поштомат громади»

    Person(resident, "Мешканець", "Забирає посилку за кодом")
    Person(delivery, "Служба доставки", "Реєструє посилки")
    Person(staff, "Персонал громади", "Переглядає список посилок")
    Person(monitor, "Моніторинг", "Перевіряє стан сервісу")

    System_Ext(sms, "SMS-провайдер", "Надсилає SMS через HTTP API")

    System_Boundary(system, "Поштомат громади — одна VM") {
        Container(api, "API", "Node.js, HTTP", "Реєструє посилки (POST /parcels), приймає код отримання (POST /pickup), показує список (GET /admin/parcels) і стан (GET /health)")
        Container(worker, "Worker", "Node.js", "Кожні 30 секунд знаходить посилки без сповіщення та запускає надсилання SMS із 4-значним кодом")
        Container(expirer, "Нічна задача", "Node.js, запланований запуск", "Позначає посилки, що зберігаються понад 72 години, як повернені")
        ContainerDb(db, "База даних", "PostgreSQL", "Зберігає посилки, події та сесії персоналу")
    }

    Rel(delivery, api, "Реєструє посилку", "HTTP")
    Rel(resident, api, "Надсилає код отримання", "HTTP")
    Rel(staff, api, "Запитує список посилок", "HTTP")
    Rel(monitor, api, "Перевіряє стан", "HTTP")
    Rel(api, db, "Читає та змінює дані")
    Rel(worker, db, "Знаходить посилки без сповіщення")
    Rel(worker, sms, "Замовляє SMS через HTTP API")
    Rel(expirer, db, "Позначає прострочені посилки поверненими")
```

**Припущення:** екран поштомата представлено як клієнт мешканця, що взаємодіє з API; окремий екран чи пристрій у документах не описаний. Нічну задачу показано як окремий контейнер із запланованим запуском. PostgreSQL — виробнича база даних; у локальному середовищі сховище працює в пам’яті.
